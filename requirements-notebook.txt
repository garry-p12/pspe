# Light dependencies for Colab / Kaggle that the base image may not ship.
#
#     !pip uninstall -q -y torchao          # avoid a transformers<->torchao clash
#     !pip install -q -r requirements-notebook.txt
#     !pip install -q -e ".[llm]"           # HF stack via >= floors (no reinstalls)
#
# DESIGN: lower bounds only, chosen LOW enough that Colab's preinstalled stack
# already satisfies them — so pip does NOT reinstall torch, numpy, scipy, pandas,
# matplotlib, transformers, or anything CUDA-linked. Exact `==` pins are what
# break Colab: they drag the resolver into upgrading torch / torchao / CUDA
# wheels against the wrong runtime. Never pin those here.
#
# Deliberately NOT listed (Colab already provides compatible versions; let them
# be): torch, numpy, scipy, pandas, matplotlib, transformers, accelerate,
# pillow, tensorboard, rich. The HF stack (transformers/peft/accelerate) is
# pulled by `pip install -e ".[llm]"` through its >= floors, which Colab meets.

gymnasium>=0.29
hydra-core>=1.3
omegaconf>=2.3

# Real data loaders (EuroSAT via the Hub, PDEBench HDF5).
datasets>=2.14
h5py>=3.8

# 4-bit quantisation on the T4. Floor, not a pin — Colab usually ships a
# compatible build already; if absent this installs a recent one.
bitsandbytes>=0.43
